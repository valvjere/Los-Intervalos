limpieza_ece_ci0115.R Lenguaje: R 
# ==============================================================================
# PROYECTO CI-0115 PROBABILIDAD Y ESTADÍSTICA - UCR
# SCRIPT DE LIMPIEZA DE DATOS (ECE 2024-2025)
# Integrantes: Ignacio, Jeremy, Lucas, Daniel
# ==============================================================================

# Cargar librerías necesarias
library(haven)      # Para leer archivos .sav de SPSS
library(tidyverse)  # Manipulación de datos (dplyr, tidyr, readr)
library(ggplot2)    # Diagnóstico visual de outliers

# ------------------------------------------------------------------------------
# PARTE 1: INGESTA Y REDUCCIÓN DE DIMENSIONALIDAD (Ignacio)
# ------------------------------------------------------------------------------
cat("--- Paso 1: Cargando y fusionando archivos SPSS ---
")
archivos_sav <- list.files("data", pattern = "\.sav$", full.names = TRUE)

# Lectura y consolidación de los 8 trimestres
df_crudo <- map_df(archivos_sav, read_sav)
n_filas_inicial <- nrow(df_crudo)
n_col_inicial <- ncol(df_crudo)
cat("Dimensiones iniciales:", n_filas_inicial, "filas y", n_col_inicial, "columnas
")

# Reducción de Dimensionalidad: Seleccionar variables críticas para el objetivo
df_reducido <- df_crudo %>%
  select(
    ID_AMO, ID_TRIMESTRE, ID_VIVIENDA, ID_HOGAR, ID_LINEA,
    Sexo, Edad, Condicion_actividad,
    Nivel_educativo, Amos_educacion, Educacion_titulo,
    Ingreso_principal, Ingreso_total, Salario_bruto,
    Horas_hab_principal
  )

# Filtrado de duplicados
df_sin_duplicados <- df_reducido %>%
  distinct(ID_AMO, ID_TRIMESTRE, ID_VIVIENDA, ID_HOGAR, ID_LINEA, .keep_all = TRUE)
n_duplicados <- nrow(df_reducido) - nrow(df_sin_duplicados)
cat("Duplicados eliminados:", n_duplicados, "
")

# ------------------------------------------------------------------------------
# PARTE 2: FILTRADO POBLACIONAL Y TIPOS DE DATOS (Jeremy)
# ------------------------------------------------------------------------------
cat("--- Paso 2: Filtrado de población objetivo ---
")
# Delimitación: Personas ocupadas (1), entre 18 y 65 años, Sexo Hombre (1) o Mujer (2)
df_filtrado <- df_sin_duplicados %>%
  filter(
    Condicion_actividad == 1,
    Edad >= 18 & Edad <= 65,
    Sexo %in% c(1, 2)
  ) %>%
  mutate(
    Edad = as.integer(Edad),
    Sexo = as.factor(Sexo),
    Ingreso_principal = as.numeric(Ingreso_principal),
    Salario_bruto = as.numeric(Salario_bruto)
  )

# Incoherencias Lógicas:
# a) Salario neto no puede superar el bruto (tras retenciones sociales)
# b) Menores de 20 años con maestría/doctorado
df_consistente <- df_filtrado %>%
  filter(
    !(Ingreso_principal > Salario_bruto & !is.na(Salario_bruto) & Salario_bruto > 0),
    !(Edad < 21 & Nivel_educativo >= 8) # Nivel 8 = Posgrado
  )

# ------------------------------------------------------------------------------
# PARTE 3: FALTANTES Y CODIFICACIÓN CATEGÓRICA (Lucas)
# ------------------------------------------------------------------------------
cat("--- Paso 3: Tratamiento de faltantes y codificación ---
")
# En ECE: 99999999 = No especificado; 0 = Sin ingreso reportado
df_faltantes <- df_consistente %>%
  mutate(
    Ingreso_principal = ifelse(Ingreso_principal %in% c(99999999, 0), NA, Ingreso_principal)
  ) %>%
  filter(!is.na(Ingreso_principal)) # Listwise deletion justificada en la variable respuesta

# Codificación Categórica de Nivel Educativo y Sexo
df_codificado <- df_faltantes %>%
  mutate(
    Sexo = factor(Sexo, levels = c(1, 2), labels = c("Hombre", "Mujer")),
    Nivel_educativo_cat = factor(
      case_when(
        Nivel_educativo %in% c(0, 1) ~ "Sin instrucción / Primaria incompleta",
        Nivel_educativo == 2        ~ "Primaria completa",
        Nivel_educativo == 3        ~ "Secundaria incompleta",
        Nivel_educativo == 4        ~ "Secundaria completa",
        Nivel_educativo == 5        ~ "Superior no universitaria",
        Nivel_educativo == 6        ~ "Superior universitaria sin título",
        Nivel_educativo == 7        ~ "Superior universitaria con título",
        Nivel_educativo >= 8        ~ "Posgrado",
        TRUE                        ~ "Otro / Desconocido"
      ),
      levels = c(
        "Sin instrucción / Primaria incompleta", "Primaria completa",
        "Secundaria incompleta", "Secundaria completa",
        "Superior no universitaria", "Superior universitaria sin título",
        "Superior universitaria con título", "Posgrado"
      ),
      ordered = TRUE
    ),
    Grupo_edad = cut(
      Edad,
      breaks = c(17, 24, 34, 49, 65),
      labels = c("18-24", "25-34", "35-49", "50-65")
    )
  )

# ------------------------------------------------------------------------------
# PARTE 4: ATÍPICOS, ESCALADO Y VALIDACIÓN DE DOMINIO (Daniel)
# ------------------------------------------------------------------------------
cat("--- Paso 4: Tratamiento de atípicos y normalización ---
")
# Detección IQR de Outliers en Ingreso
Q1 <- quantile(df_codificado$Ingreso_principal, 0.25, na.rm = TRUE)
Q3 <- quantile(df_codificado$Ingreso_principal, 0.75, na.rm = TRUE)
IQR_val <- Q3 - Q1
lim_inf <- max(Q1 - 1.5 * IQR_val, 50000) # Salario mínimo legal de subsistencia
lim_sup <- Q3 + 3.0 * IQR_val             # Outliers severos

# Filtro de Outliers extremos
df_limpio <- df_codificado %>%
  filter(Ingreso_principal >= lim_inf & Ingreso_principal <= lim_sup) %>%
  mutate(
    log_ingreso = log(Ingreso_principal),
    z_ingreso = as.numeric(scale(Ingreso_principal))
  )

# Métricas finales
cat("=====================================================
")
cat("RESULTADO FINAL TRAS LA LIMPIEZA:
")
cat("Filas finales:", nrow(df_limpio), "(Retención:", round(nrow(df_limpio)/n_filas_inicial*100, 2), "%)
")
cat("Columnas finales:", ncol(df_limpio), "
")
cat("Media salarial neta: ₡", round(mean(df_limpio$Ingreso_principal)), "
")
cat("Mediana salarial neta: ₡", round(median(df_limpio$Ingreso_principal)), "
")
cat("=====================================================
")

# Guardar dataset limpio resultante
write_csv(df_limpio, "dataset_limpio_ece_2024_2025.csv")
cat("Archivo 'dataset_limpio_ece_2024_2025.csv' generado con éxito.
")

# ==============================================================================

import glob
import numpy as np
import pandas as pd
import pyreadstat

# ------------------------------------------------------------------------------
# PARTE 1: INGESTA Y REDUCCIÓN DE DIMENSIONALIDAD (Ignacio)
# ------------------------------------------------------------------------------
print("--- Paso 1: Ingesta de archivos .sav ---")
files = sorted(glob.glob("data/*.sav"))
dfs = []
for f in files:
    data, meta = pyreadstat.read_sav(f)
    dfs.append(data)

df = pd.concat(dfs, ignore_index=True)
n_inicial_filas, n_inicial_cols = df.shape
print(f"Dimensiones iniciales: {n_inicial_filas} filas, {n_inicial_cols} columnas")

# Reducción de dimensionalidad
cols_keep = [
    'ID_AMO', 'ID_TRIMESTRE', 'ID_VIVIENDA', 'ID_HOGAR', 'ID_LINEA',
    'Sexo', 'Edad', 'Condicion_actividad',
    'Nivel_educativo', 'Amos_educacion', 'Educacion_titulo',
    'Ingreso_principal', 'Ingreso_total', 'Salario_bruto',
    'Horas_hab_principal'
]
df = df[cols_keep].copy()

# Eliminar duplicados
n_antes_dup = len(df)
df.drop_duplicates(subset=['ID_AMO', 'ID_TRIMESTRE', 'ID_VIVIENDA', 'ID_HOGAR', 'ID_LINEA'], inplace=True)
print(f"Duplicados eliminados: {n_antes_dup - len(df)}")

# ------------------------------------------------------------------------------
# PARTE 2: FILTRADO POBLACIONAL Y TIPOS DE DATOS (Jeremy)
# ------------------------------------------------------------------------------
print("--- Paso 2: Filtrado poblacional ---")
# Filtros: Ocupados (1), Edad entre 18 y 65, Sexo Hombre (1) o Mujer (2)
df = df[(df['Condicion_actividad'] == 1) & 
        (df['Edad'] >= 18) & (df['Edad'] <= 65) & 
        (df['Sexo'].isin([1, 2]))].copy()

# Tipos de datos
df['Edad'] = df['Edad'].astype(int)
df['Ingreso_principal'] = pd.to_numeric(df['Ingreso_principal'], errors='coerce')
df['Salario_bruto'] = pd.to_numeric(df['Salario_bruto'], errors='coerce')

# Incoherencias lógicas
mask_incoherente = (df['Ingreso_principal'] > df['Salario_bruto']) & (df['Salario_bruto'] > 0)
df = df[~mask_incoherente].copy()

# ------------------------------------------------------------------------------
# PARTE 3: FALTANTES Y CODIFICACIÓN CATEGÓRICA (Lucas)
# ------------------------------------------------------------------------------
print("--- Paso 3: Faltantes y codificación ---")
# Reemplazar códigos de no respuesta de la ECE
df['Ingreso_principal'] = df['Ingreso_principal'].replace([99999999, 0], np.nan)
df.dropna(subset=['Ingreso_principal'], inplace=True)

# Codificación de variables
df['Sexo'] = df['Sexo'].map({1: 'Hombre', 2: 'Mujer'}).astype('category')

edu_map = {
    0: 'Sin instrucción / Primaria incompleta',
    1: 'Sin instrucción / Primaria incompleta',
    2: 'Primaria completa',
    3: 'Secundaria incompleta',
    4: 'Secundaria completa',
    5: 'Superior no universitaria',
    6: 'Superior universitaria sin título',
    7: 'Superior universitaria con título',
    8: 'Posgrado'
}
df['Nivel_educativo_cat'] = df['Nivel_educativo'].map(edu_map).fillna('Otro')

df['Grupo_edad'] = pd.cut(
    df['Edad'], 
    bins=[17, 24, 34, 49, 65], 
    labels=['18-24', '25-34', '35-49', '50-65']
)

# ------------------------------------------------------------------------------
# PARTE 4: ATÍPICOS, ESCALADO Y VALIDACIÓN (Daniel)
# ------------------------------------------------------------------------------
print("--- Paso 4: Tratamiento de atípicos y escalado ---")
Q1 = df['Ingreso_principal'].quantile(0.25)
Q3 = df['Ingreso_principal'].quantile(0.75)
IQR = Q3 - Q1
lim_inf = max(Q1 - 1.5 * IQR, 50000)
lim_sup = Q3 + 3.0 * IQR

df_limpio = df[(df['Ingreso_principal'] >= lim_inf) & (df['Ingreso_principal'] <= lim_sup)].copy()

# Escalado y normalización
df_limpio['log_ingreso'] = np.log(df_limpio['Ingreso_principal'])
df_limpio['z_ingreso'] = (df_limpio['Ingreso_principal'] - df_limpio['Ingreso_principal'].mean()) / df_limpio['Ingreso_principal'].std()

print("=====================================================")
print(f"Dimensiones finales: {df_limpio.shape}")
print(f"Retención muestral: {len(df_limpio) / n_inicial_filas * 100:.2f}%")
print(f"Mediana ingreso neto: ₡{df_limpio['Ingreso_principal'].median():,.0f}")
print("=====================================================")

df_limpio.to_csv("dataset_limpio_ece_2024_2025.csv", index=False)
